/*
|--------------------------------------------------------------------------
| Shared authenticated routes (all roles)
|--------------------------------------------------------------------------
*/
Route::middleware(['auth'])->group(function () {

    // Dashboard dispatcher — sends each user to their own dashboard
    Route::get('/dashboard', function () {
        return match (auth()->user()->role) {
            'admin'   => redirect()->route('admin.dashboard'),
            'cashier' => redirect()->route('cashier.dashboard'),
            'stocker' => redirect()->route('stocker.dashboard'),
            default   => abort(403),
        };
    })->name('dashboard');

    // Shared profile routes (Breeze)
    Route::get('/profile',    [ProfileController::class, 'edit'])->name('profile.edit');
    Route::patch('/profile',  [ProfileController::class, 'update'])->name('profile.update');
    Route::delete('/profile', [ProfileController::class, 'destroy'])->name('profile.destroy');

    // Change password (separate page — Tier 2 requirement)
    Route::get('/password',   [ProfileController::class, 'editPassword'])->name('password.edit');
    Route::put('/password',   [ProfileController::class, 'updatePassword'])->name('password.update.custom');
});

===== FILE: bootstrap\app.php =====

<?php

use Illuminate\Foundation\Application;
use Illuminate\Foundation\Configuration\Exceptions;
use Illuminate\Foundation\Configuration\Middleware;

return Application::configure(basePath: dirname(__DIR__))
    ->withRouting(
        web: __DIR__.'/../routes/web.php',
        commands: __DIR__.'/../routes/console.php',
        health: '/up',
    )
    ->withMiddleware(function (Middleware $middleware) {

        // Register middleware aliases so routes can use them by short name
        $middleware->alias([
            'role'           => \App\Http\Middleware\RoleMiddleware::class,
            'session.timeout' => \App\Http\Middleware\SessionTimeout::class,
        ]);

        // Append session timeout to every web request
        $middleware->web(append: [
            \App\Http\Middleware\SessionTimeout::class,
        ]);
    })
    ->withExceptions(function (Exceptions $exceptions) {
        //
    })->create();

<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\AdjustmentRequest;
use App\Models\AuditLog;
use App\Models\StockLog;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\View\View;

class AdjustmentController extends Controller
{
    public function index(Request $request): View
    {
        $query = AdjustmentRequest::with(['product', 'requestedBy']);

        if ($status = $request->get('status')) {
            $query->where('status', $status);
        }

        $adjustments = $query->latest()->paginate(15)->withQueryString();

        return view('admin.adjustments.index', compact('adjustments'));
    }

    public function show(AdjustmentRequest $adjustment): View
    {
        $adjustment->load(['product', 'requestedBy', 'reviewedBy']);

        return view('admin.adjustments.show', compact('adjustment'));
    }

    public function approve(Request $request, AdjustmentRequest $adjustment): RedirectResponse
    {
        if (! $adjustment->isPending()) {
            return back()->with('error', 'This request has already been processed.');
        }

        DB::transaction(function () use ($request, $adjustment) {
            $product = $adjustment->product;

            $before = $product->stock_qty;
            $product->update(['stock_qty' => $adjustment->new_qty]);
            $after = $product->fresh()->stock_qty;

            StockLog::create([
                'product_id'   => $product->id,
                'user_id'      => auth()->id(),
                'type'         => 'adjustment',
                'qty'          => $adjustment->difference,
                'stock_before' => $before,
                'stock_after'  => $after,
                'reason'       => $adjustment->reason,
            ]);

            $adjustment->update([
                'status'      => 'approved',
                'reviewed_by' => auth()->id(),
                'reviewed_at' => now(),
            ]);

            AuditLog::create([
                'user_id'    => auth()->id(),
                'action'     => 'adjustment.approved',
                'model_type' => AdjustmentRequest::class,
                'model_id'   => $adjustment->id,
                'ip_address' => $request->ip(),
                'user_agent' => $request->userAgent(),
            ]);
        });

        return redirect()->route('admin.adjustments.index')
            ->with('success', 'Adjustment approved. Stock has been updated.');
    }

    public function reject(Request $request, AdjustmentRequest $adjustment): RedirectResponse
    {
        if (! $adjustment->isPending()) {
            return back()->with('error', 'This request has already been processed.');
        }

        $validated = $request->validate([
            'rejection_reason' => ['required', 'string', 'max:500'],
        ]);

        $adjustment->update([
            'status'           => 'rejected',
            'reviewed_by'      => auth()->id(),
            'reviewed_at'      => now(),
            'rejection_reason' => $validated['rejection_reason'],
        ]);

        AuditLog::create([
            'user_id'    => auth()->id(),
            'action'     => 'adjustment.rejected',
            'model_type' => AdjustmentRequest::class,
            'model_id'   => $adjustment->id,
            'new_values' => ['reason' => $validated['rejection_reason']],
            'ip_address' => $request->ip(),
            'user_agent' => $request->userAgent(),
        ]);

        return redirect()->route('admin.adjustments.index')
            ->with('success', 'Adjustment rejected.');
    }
}

namespace App\Http\Controllers\Admin;

class ExportController extends Controller

    /**
     * Print-ready sales report.
     */
    public function sales(Request $request): View
    {
        $from = $request->get('from', now()->startOfMonth()->toDateString());
        $to   = $request->get('to', now()->toDateString());

        $sales = Transaction::with('cashier')
            ->completed()
            ->whereBetween('created_at', [$from . ' 00:00:00', $to . ' 23:59:59'])
            ->orderBy('created_at')
            ->get();

        return view('reports.print-sales', compact('sales', 'from', 'to'));
    }

    /**
     * Print-ready inventory summary.
     */
    public function inventory(): View
    {
        $products = Product::with(['category', 'supplier'])->orderBy('name')->get();
        $totalValue = $products->sum(fn($p) => $p->stock_qty * $p->cost);

        return view('reports.print-inventory', compact('products', 'totalValue'));
    }

    /**
     * Print-ready staff activity report.
     */
    public function staff(Request $request): View
    {
        $from = $request->get('from', now()->startOfMonth()->toDateString());
        $to   = $request->get('to', now()->toDateString());

        $staff = User::whereIn('role', ['cashier', 'stocker'])
            ->withCount(['transactions' => function ($q) use ($from, $to) {
                $q->whereBetween('created_at', [$from . ' 00:00:00', $to . ' 23:59:59']);
            }])
            ->withSum(['transactions' => function ($q) use ($from, $to) {
                $q->whereBetween('created_at', [$from . ' 00:00:00', $to . ' 23:59:59']);
            }], 'total')
            ->get();

        return view('reports.print-staff', compact('staff', 'from', 'to'));
    }

    /**
     * Print-ready financial report.
     */
    public function financial(Request $request): View
    {
        $from = $request->get('from', now()->startOfMonth()->toDateString());
        $to   = $request->get('to', now()->toDateString());

        $base = Transaction::completed()
            ->whereBetween('created_at', [$from . ' 00:00:00', $to . ' 23:59:59']);

        $revenue    = (clone $base)->sum('total');
        $count      = (clone $base)->count();
        $cashTotal  = (clone $base)->where('payment_method', 'cash')->sum('total');
        $gcashTotal = (clone $base)->where('payment_method', 'gcash')->sum('total');

        // Gross profit
        $profit = \App\Models\TransactionItem::whereHas('transaction', function ($q) use ($from, $to) {
                $q->whereBetween('created_at', [$from . ' 00:00:00', $to . ' 23:59:59'])
                  ->where('status', 'completed');
            })
            ->selectRaw('SUM((unit_price - unit_cost) * qty) as total_profit')
            ->value('total_profit') ?? 0;

        return view('reports.print-financial', compact(
            'from', 'to', 'revenue', 'count', 'cashTotal', 'gcashTotal', 'profit'
        ));
    }
}

            ->whereBetween('created_at', [$from . ' 00:00:00', $to . ' 23:59:59']);

<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\AuditLog;
use App\Models\VoidRequest;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\View\View;

class VoidController extends Controller
{
    public function index(Request $request): View
    {
        $query = VoidRequest::with(['transaction.cashier', 'requestedBy']);

        if ($status = $request->get('status')) {
            $query->where('status', $status);
        }

        $voids = $query->latest()->paginate(15)->withQueryString();

        return view('admin.voids.index', compact('voids'));
    }

    public function show(VoidRequest $void): View
    {
        $void->load(['transaction.items.product', 'transaction.cashier', 'requestedBy', 'reviewedBy']);

        return view('admin.voids.show', compact('void'));
    }

    public function approve(Request $request, VoidRequest $void): RedirectResponse
    {
        if (! $void->isPending()) {
            return back()->with('error', 'This request has already been processed.');
        }

        DB::transaction(function () use ($request, $void) {
            $void->update([
                'status'      => 'approved',
                'reviewed_by' => auth()->id(),
                'reviewed_at' => now(),
            ]);

            // Void the transaction
            $void->transaction->update(['status' => 'voided']);

            // Restore stock
            foreach ($void->transaction->items as $item) {
                $product = $item->product;
                if (! $product) continue;

                $before = $product->stock_qty;
                $product->increment('stock_qty', $item->qty);
                $after = $product->fresh()->stock_qty;

                \App\Models\StockLog::create([
                    'product_id'   => $product->id,
                    'user_id'      => auth()->id(),
                    'type'         => 'void_restock',
                    'qty'          => $item->qty,
                    'stock_before' => $before,
                    'stock_after'  => $after,
                    'reason'       => 'Voided transaction #' . $void->transaction->reference_no,
                ]);
            }

            AuditLog::create([
                'user_id'    => auth()->id(),
                'action'     => 'void.approved',
                'model_type' => VoidRequest::class,
                'model_id'   => $void->id,
                'ip_address' => $request->ip(),
                'user_agent' => $request->userAgent(),
            ]);
        });

        return redirect()->route('admin.voids.index')
            ->with('success', 'Void approved. Stock has been restored.');
    }

    public function reject(Request $request, VoidRequest $void): RedirectResponse
    {
        if (! $void->isPending()) {
            return back()->with('error', 'This request has already been processed.');
        }

        $validated = $request->validate([
            'rejection_reason' => ['required', 'string', 'max:500'],
        ]);

        $void->update([
            'status'           => 'rejected',
            'reviewed_by'      => auth()->id(),
            'reviewed_at'      => now(),
            'rejection_reason' => $validated['rejection_reason'],
        ]);

        AuditLog::create([
            'user_id'    => auth()->id(),
            'action'     => 'void.rejected',
            'model_type' => VoidRequest::class,
            'model_id'   => $void->id,
            'new_values' => ['reason' => $validated['rejection_reason']],
            'ip_address' => $request->ip(),
            'user_agent' => $request->userAgent(),
        ]);

        return redirect()->route('admin.voids.index')
            ->with('success', 'Void request rejected.');
    }
}

use Illuminate\Http\Request;

use Illuminate\Support\Str;
use Illuminate\Validation\Rules;
use Illuminate\Validation\ValidationException;

{
    /**
     * Display the password reset view.
     */
    public function create(Request $request): View
    {
        return view('auth.reset-password', ['request' => $request]);
    }

    /**
     * Handle an incoming new password request.
     *
     * @throws ValidationException
     */
    public function store(Request $request): RedirectResponse
    {
        $request->validate([
            'token' => ['required'],
            'email' => ['required', 'email'],
            'password' => ['required', 'confirmed', Rules\Password::defaults()],
        ]);

        // Here we will attempt to reset the user's password. If it is successful we
        // will update the password on an actual user model and persist it to the
        // database. Otherwise we will parse the error and return the response.
        $status = Password::reset(
            $request->only('email', 'password', 'password_confirmation', 'token'),
            function (User $user) use ($request) {
                $user->forceFill([
                    'password' => Hash::make($request->password),
                    'remember_token' => Str::random(60),
                ])->save();

                event(new PasswordReset($user));
            }
        );

        // If the password was successfully reset, we will redirect the user back to
        // the application's home authenticated view. If there is an error we can
        // redirect them back to where they came from with their error message.
        return $status == Password::PASSWORD_RESET
                    ? redirect()->route('login')->with('status', __($status))
                    : back()->withInput($request->only('email'))
                        ->withErrors(['email' => __($status)]);
    }
}

     * Display the registration view.

     * Handle an incoming registration request.
     *
     * @throws ValidationException
     */
    public function store(Request $request): RedirectResponse
    {
        $request->validate([
            'name' => ['required', 'string', 'max:255'],
            'email' => ['required', 'string', 'lowercase', 'email', 'max:255', 'unique:'.User::class],
            'password' => ['required', 'confirmed', Rules\Password::defaults()],
        ]);

        $user = User::create([
            'name' => $request->name,
            'email' => $request->email,
            'password' => Hash::make($request->password),
        ]);

        event(new Registered($user));

        Auth::login($user);

        return redirect(route('dashboard', absolute: false));
    }
}

<?php

namespace App\Http\Controllers\Cashier;

use App\Http\Controllers\Controller;
use App\Http\Requests\CheckoutRequest;
use App\Models\AuditLog;
use App\Models\Product;
use App\Models\StockLog;
use App\Models\Transaction;
use App\Models\TransactionItem;
use App\Services\CartService;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\View\View;

class POSController extends Controller
{
    public function __construct(protected CartService $cart) {}

    /**
     * Show the POS screen.
     */
    public function index(Request $request): View
    {
        $search = $request->get('q');

        $products = Product::active()
            ->where('stock_qty', '>', 0)
            ->when($search, function ($q) use ($search) {
                $q->where(function ($sub) use ($search) {
                    $sub->where('name', 'like', "%{$search}%")
                        ->orWhere('sku', 'like', "%{$search}%");
                });
            })
            ->orderBy('name')
            ->limit(24)
            ->get();

        return view('cashier.pos', [
            'products' => $products,
            'cart'     => $this->cart->detailed(),
            'subtotal' => $this->cart->subtotal(),
            'total'    => $this->cart->total(),
            'search'   => $search,
        ]);
    }

    /**
     * Add a product to the cart.
     */
    public function add(Request $request, Product $product): RedirectResponse
    {
        if ($product->stock_qty <= 0) {
            return back()->with('error', 'Product is out of stock.');
        }

        $qty = (int) $request->input('qty', 1);

        $alreadyInCart = $this->cart->qtyOf($product->id);

        if ($alreadyInCart + $qty > $product->stock_qty) {
            return back()->with('error', "Only {$product->stock_qty} in stock.");
        }

        $this->cart->add($product->id, $qty);

        return back()->with('success', "{$product->name} added to cart.");
    }

    /**
     * Update quantity of a cart item.
     */
    public function update(Request $request, Product $product): RedirectResponse
    {
        $qty = (int) $request->input('qty', 1);

        if ($qty > $product->stock_qty) {
            return back()->with('error', "Only {$product->stock_qty} in stock.");
        }

        $this->cart->update($product->id, $qty);

        return back();
    }

    /**
     * Remove a product from the cart.
     */
    public function remove(Product $product): RedirectResponse
    {
        $this->cart->remove($product->id);

        return back()->with('success', 'Item removed from cart.');
    }

    /**
     * Clear the entire cart.
     */
    public function clear(): RedirectResponse
    {
        $this->cart->clear();

        return back()->with('success', 'Cart cleared.');
    }

    /**
     * Process the checkout.
     */
    public function checkout(CheckoutRequest $request): RedirectResponse
    {
        $cart = $this->cart->detailed();

        if (empty($cart)) {
            return back()->with('error', 'Cart is empty.');
        }

        $subtotal = $this->cart->subtotal();
        $discount = 0;
        $total    = $subtotal;

        // Validate payment amount
        $amountPaid = (float) $request->input('amount_paid');

        if ($amountPaid < $total) {
            return back()->with('error', 'Amount paid is less than total.');
        }

        $transaction = DB::transaction(function () use ($request, $cart, $subtotal, $discount, $total, $amountPaid) {

            // Generate reference number
            $referenceNo = 'CT-' . now()->format('Ymd') . '-' . str_pad(
                Transaction::whereDate('created_at', today())->count() + 1,
                4,
                '0',
                STR_PAD_LEFT
            );

            $transaction = Transaction::create([
                'reference_no'   => $referenceNo,
                'cashier_id'     => auth()->id(),
                'subtotal'       => $subtotal,
                'discount'       => $discount,
                'total'          => $total,
                'payment_method' => $request->input('payment_method'),
                'amount_paid'    => $amountPaid,
                'change_due'     => $amountPaid - $total,
                'status'         => 'completed',
                'notes'          => $request->input('notes'),
            ]);

            // Save line items + deduct stock
            foreach ($cart as $row) {
                $product = $row['product'];

                TransactionItem::create([
                    'transaction_id' => $transaction->id,
                    'product_id'     => $product->id,
                    'product_name'   => $product->name,
                    'qty'            => $row['qty'],
                    'unit_price'     => $product->price,
                    'unit_cost'      => $product->cost,
                    'subtotal'       => $row['subtotal'],
                ]);

                $before = $product->stock_qty;
                $product->decrement('stock_qty', $row['qty']);
                $after = $product->fresh()->stock_qty;

                StockLog::create([
                    'product_id'   => $product->id,
                    'user_id'      => auth()->id(),
                    'type'         => 'sale_deduction',
                    'qty'          => -$row['qty'],
                    'stock_before' => $before,
                    'stock_after'  => $after,
                    'reason'       => 'Sale ' . $referenceNo,
                ]);
            }

            AuditLog::create([
                'user_id'    => auth()->id(),
                'action'     => 'sale.completed',
                'model_type' => Transaction::class,
                'model_id'   => $transaction->id,
                'new_values' => ['reference_no' => $referenceNo, 'total' => $total],
                'ip_address' => $request->ip(),
                'user_agent' => $request->userAgent(),
            ]);

            return $transaction;
        });

        $this->cart->clear();

        return redirect()->route('cashier.receipt', $transaction)
            ->with('success', 'Sale completed.');
    }
}

<?php

namespace App\Http\Controllers\Cashier;

use App\Http\Controllers\Controller;
use App\Http\Requests\ReturnRequest;
use App\Models\AuditLog;
use App\Models\ReturnItem;
use App\Models\StockLog;
use App\Models\Transaction;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\View\View;

class ReturnController extends Controller
{
    public function index(Request $request): View
    {
        $transactions = Transaction::with(['items', 'cashier', 'returns'])
            ->where('status', '!=', 'voided')
            ->where('created_at', '>=', now()->subDays(7))
            ->latest()
            ->paginate(15);

        return view('cashier.returns', compact('transactions'));
    }

    public function create(Transaction $transaction): View
    {
        $hoursSince = $transaction->created_at->diffInHours(now());

        if ($hoursSince > 24) {
            abort(403, 'Return window (24 hours) has expired.');
        }

        if ($transaction->status === 'voided' || $transaction->status === 'returned') {
            abort(403, 'This transaction is no longer eligible for returns.');
        }

        $transaction->load(['items.product', 'cashier']);

        return view('cashier.returns', [
            'transactions' => collect(),
            'transaction'  => $transaction,
        ]);
    }

    public function store(ReturnRequest $request, Transaction $transaction): RedirectResponse
    {
        if ($transaction->created_at->diffInHours(now()) > 24) {
            return back()->with('error', 'Return window (24 hours) has expired.');
        }

        if ($transaction->status === 'voided' || $transaction->status === 'returned') {
            return back()->with('error', 'This transaction is no longer eligible for returns.');
        }

        $validated = $request->validated();

        DB::transaction(function () use ($request, $transaction, $validated) {
            $totalRefund = 0;

            foreach ($validated['items'] as $itemData) {
                // Skip items that were not checked (qty = 0)
                if ($itemData['qty'] < 1) {
                    continue;
                }

                $item = $transaction->items()->findOrFail($itemData['id']);
                $product = $item->product;

                $refundAmount = $item->unit_price * $itemData['qty'];
                $totalRefund += $refundAmount;

                ReturnItem::create([
                    'transaction_id' => $transaction->id,
                    'product_id'     => $product->id,
                    'processed_by'   => auth()->id(),
                    'qty'            => $itemData['qty'],
                    'refund_amount'  => $refundAmount,
                    'type'           => $validated['type'],
                    'reason'         => $validated['reason'],
                ]);

                $before = $product->stock_qty;
                $product->increment('stock_qty', $itemData['qty']);
                $after = $product->fresh()->stock_qty;

                StockLog::create([
                    'product_id'   => $product->id,
                    'user_id'      => auth()->id(),
                    'type'         => 'return_restock',
                    'qty'          => $itemData['qty'],
                    'stock_before' => $before,
                    'stock_after'  => $after,
                    'reason'       => 'Return on ' . $transaction->reference_no,
                ]);
            }

            // UPDATED: If ANY item was returned, mark the whole transaction as returned
            if ($totalRefund > 0) {
                $transaction->update(['status' => 'returned']);
            }

            AuditLog::create([
                'user_id'    => auth()->id(),
                'action'     => 'return.processed',
                'model_type' => ReturnItem::class,
                'new_values' => [
                    'transaction'  => $transaction->reference_no,
                    'total_refund' => $totalRefund,
                ],
                'ip_address' => $request->ip(),
                'user_agent' => $request->userAgent(),
            ]);
        });

        return redirect()->route('cashier.returns.index')
            ->with('success', 'Return processed. Items restocked.');
    }
}

    /**
     * List this cashier's void requests.
     */
    public function index(): View
    {
        $voids = VoidRequest::with('transaction')
            ->where('requested_by', auth()->id())
            ->latest()
            ->paginate(15);

        return view('cashier.voids.index', compact('voids'));
    }

    /**
     * Show the void request form.
     */
    public function create(Transaction $transaction): View
    {
        if ($transaction->cashier_id !== auth()->id()) {
            abort(403);
        }

        if ($transaction->status !== 'completed') {
            abort(403, 'Only completed transactions can be voided.');
        }

        if ($transaction->voidRequest()->exists()) {
            abort(403, 'A void request already exists for this transaction.');
        }

        $transaction->load('items');

        return view('cashier.voids.create', compact('transaction'));
    }

    /**
     * Submit a void request.
     */
    public function store(VoidRequestRequest $request, Transaction $transaction): RedirectResponse
    {
        if ($transaction->cashier_id !== auth()->id()) {
            abort(403);
        }

        if ($transaction->status !== 'completed') {
            return back()->with('error', 'Only completed transactions can be voided.');
        }

        if ($transaction->voidRequest()->exists()) {
            return back()->with('error', 'A void request already exists.');
        }

        $voidRequest = VoidRequest::create([
            'transaction_id' => $transaction->id,
            'requested_by'   => auth()->id(),
            'reason'         => $request->validated()['reason'],
            'status'         => 'pending',
        ]);

        AuditLog::create([
            'user_id'    => auth()->id(),
            'action'     => 'void.requested',
            'model_type' => VoidRequest::class,

                'transaction' => $transaction->reference_no,
                'reason'      => $request->validated()['reason'],
            ],
            'ip_address' => $request->ip(),
            'user_agent' => $request->userAgent(),
        ]);

        return redirect()->route('cashier.voids.index')
            ->with('success', 'Void request submitted. Awaiting admin approval.');
    }

        return view('stocker.stock-in', compact('products', 'suppliers', 'recentLogs'));
    }

    /**
     * Record a stock-in entry.
     */
    public function store(StockInRequest $request): RedirectResponse
    {
        $validated = $request->validated();

        DB::transaction(function () use ($request, $validated) {
            $product = Product::findOrFail($validated['product_id']);

            $before = $product->stock_qty;
            $product->increment('stock_qty', $validated['qty']);
            $after = $product->fresh()->stock_qty;

            // Optionally update cost if provided
            if (! empty($validated['unit_cost'])) {
                $product->update(['cost' => $validated['unit_cost']]);
            }

            StockLog::create([
                'product_id'   => $product->id,
                'supplier_id'  => $validated['supplier_id'] ?? null,
                'user_id'      => auth()->id(),
                'type'         => 'stock_in',
                'qty'          => $validated['qty'],
                'stock_before' => $before,
                'stock_after'  => $after,
                'unit_cost'    => $validated['unit_cost'] ?? null,
                'reason'       => $validated['notes'] ?? null,
            ]);

            AuditLog::create([
                'user_id'    => auth()->id(),
                'action'     => 'stock.in',
                'model_type' => Product::class,
                'model_id'   => $product->id,
                'new_values' => [
                    'product'      => $product->name,
                    'qty_added'    => $validated['qty'],
                    'stock_before' => $before,
                    'stock_after'  => $after,
                ],
                'ip_address' => $request->ip(),
                'user_agent' => $request->userAgent(),
            ]);
        });

        return redirect()->route('stocker.stock-in.index')
            ->with('success', 'Stock-in recorded successfully.');
    }
}

        ]);

    }

<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class RoleMiddleware
{
    /**
     * Handle an incoming request.
     *
     * Usage in routes:
     *   ->middleware('role:admin')
     *   ->middleware('role:admin,cashier')
     *   ->middleware('role:stocker')
     */
    public function handle(Request $request, Closure $next, string ...$roles): Response
    {
        // 1. Must be logged in
        if (! $request->user()) {
            return redirect()->route('login');
        }

        // 2. Account must be active
        if (! $request->user()->is_active) {
            auth()->logout();
            return redirect()->route('login')
                ->withErrors(['email' => 'Your account has been deactivated.']);
        }

        // 3. Role must match one of the allowed roles
        if (! in_array($request->user()->role, $roles, true)) {
            abort(403, 'You do not have permission to access this page.');
        }

        return $next($request);
    }
}
